const carouselState = new WeakMap();

function carouselHTML(images, alt) {
  if (!images || !images.length) return `<div class="carousel-placeholder">No image yet</div>`;
  const track = images.map((src, i) => `
    <div class="carousel__slide"><img src="${esc(src)}" alt="${esc(alt)} — view ${i+1}" loading="${i===0?'eager':'lazy'}"></div>
  `).join('');
  const dots = images.length > 1 ? `
    <div class="carousel__dots">
      ${images.map((_, i) => `<button class="dot ${i===0?'active':''}" data-idx="${i}" aria-label="Go to image ${i+1}"></button>`).join('')}
    </div>
  ` : '';
  const arrows = images.length > 1 ? `
    <button class="carousel__arrow carousel__arrow--prev" aria-label="Previous image">‹</button>
    <button class="carousel__arrow carousel__arrow--next" aria-label="Next image">›</button>
  ` : '';
  return `
    <div class="carousel" data-carousel data-count="${images.length}">
      <div class="carousel__viewport">
        <div class="carousel__track" style="transform:translateX(0%)">${track}</div>
      </div>
      ${arrows}${dots}
    </div>
  `;
}

function initCarousel(root) {
  if (!root) return;
  const track = root.querySelector('.carousel__track');
  const dots  = [...root.querySelectorAll('.dot')];
  const prev  = root.querySelector('.carousel__arrow--prev');
  const next  = root.querySelector('.carousel__arrow--next');
  const count = Number(root.dataset.count || 1);
  let idx = 0;

  function go(i) {
    idx = (i + count) % count;
    track.style.transform = `translateX(-${idx * 100}%)`;
    dots.forEach((d, j) => d.classList.toggle('active', j === idx));
  }
  if (prev) prev.addEventListener('click', () => go(idx - 1));
  if (next) next.addEventListener('click', () => go(idx + 1));
  dots.forEach(d => d.addEventListener('click', () => go(Number(d.dataset.idx))));

  // touch swipe
  let startX = null;
  root.addEventListener('touchstart', e => { startX = e.touches[0].clientX; }, { passive: true });
  root.addEventListener('touchend', e => {
    if (startX == null) return;
    const dx = e.changedTouches[0].clientX - startX;
    if (Math.abs(dx) > 40) go(idx + (dx < 0 ? 1 : -1));
    startX = null;
  });
  // keyboard
  root.setAttribute('tabindex', '0');
  root.addEventListener('keydown', e => {
    if (e.key === 'ArrowLeft') go(idx - 1);
    if (e.key === 'ArrowRight') go(idx + 1);
  });
}

function submitContact(e) {
  e.preventDefault();

function collectPageFromForm() {
  const db = loadDB();

  if (!p) return null;
  p.title = document.getElementById('pg_title').value.trim();
  p.slug = document.getElementById('pg_slug').value.trim();
  // update blocks from inputs
  document.querySelectorAll('[data-b]').forEach(el => {
    const idx = Number(el.dataset.b);
    const field = el.dataset.f;
    const value = el.value;
    const block = p.blocks[idx];
    if (!block) return;
    if (field.includes('.')) {
      const [k, j, kk] = field.split('.');
      block[k][Number(j)][kk] = value;
    } else {
      block[field] = value;
    }
  });
  return { db, p };
}

function adminVisitors(db) {
  const today = new Date(); today.setHours(0,0,0,0);
  const week = Date.now() - 7 * 24 * 60 * 60 * 1000;
  const todayCount = db.visits.filter(v => v.ts >= today.getTime()).length;
  const weekCount = db.visits.filter(v => v.ts >= week).length;
  const counts = {};
  db.visits.forEach(v => { counts[v.path] = (counts[v.path] || 0) + 1; });
  const topPaths = Object.entries(counts).sort((a, b) => b[1] - a[1]).slice(0, 10);
  return `
    <h1>Visitors</h1>
    <div class="stat-grid">
      <div class="stat"><b>${db.visits.length}</b><span>All time</span></div>
      <div class="stat"><b>${todayCount}</b><span>Today</span></div>
      <div class="stat"><b>${weekCount}</b><span>Last 7 days</span></div>
      <div class="stat"><b>${topPaths.length}</b><span>Distinct paths</span></div>
    </div>
    <h2>Top pages</h2>
    ${topPaths.length === 0 ? `<div class="admin-card empty">No visits recorded yet.</div>` :
      topPaths.map(([p, n]) => `<div class="lead-row">${esc(p)} <strong style="float:right">${n}</strong></div>`).join('')}
    <h2>Recent visits</h2>
    <div class="admin-card">
      ${[...db.visits].reverse().slice(0, 15).map(v => `
        <div class="lead-row" style="padding:.5rem .8rem;font-size:.8rem">${esc(v.path)} <small style="display:inline;margin-left:1rem">${new Date(v.ts).toLocaleTimeString()}</small></div>
      `).join('') || '<p class="empty">None yet.</p>'}
    </div>
  `;
}

function render() {
  const db = loadDB();
  const path = location.hash || '#/';
  const app = document.getElementById('app');
  let html = '';

  // reset title each render
  document.title = 'The Himalayan Tea Company — Est. 1849';

  if (path === '#/' || path === '') {
    html = viewHome(db, path);
  } else if (path === '#/products') {
    html = viewProducts(db, path);
    document.title = 'The Collection | The Himalayan Tea Company';
  } else if (path.startsWith('#/products/')) {
    const slug = decodeURIComponent(path.replace('#/products/', ''));
    html = viewProduct(db, slug, path);
  } else if (path.startsWith('#/l/')) {
    const slug = decodeURIComponent(path.replace('#/l/', ''));
    html = viewLanding(db, slug, path);
  } else if (path === '#/admin/login') {
    html = viewLogin();
    document.title = 'Admin · Login';
  } else if (path === '#/admin') {
    html = viewAdmin(db);
    document.title = 'Admin · Dashboard';
  } else {
    html = notFoundHTML(db, path);
  }

  app.innerHTML = html;
  window.scrollTo({ top: 0, behavior: 'instant' });

  // init carousel if present
  const car = app.querySelector('[data-carousel]');
  if (car) initCarousel(car);

  // record visit (not on admin)
  if (!path.startsWith('#/admin')) {
    const db2 = loadDB();
    db2.visits.push({ path, ts: Date.now() });
    if (db2.visits.length > 5000) db2.visits = db2.visits.slice(-5000);
    saveDB(db2);
  }
}
